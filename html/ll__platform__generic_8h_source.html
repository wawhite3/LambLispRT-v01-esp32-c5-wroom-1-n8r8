584class LL_String {
585public:
586
587 LL_String() { _init(); set(""); }
588 LL_String(const char *other) { _init(); set(other); }
589 LL_String(const char *other, int n) { _init(); set(other, n); }
590 LL_String(const LL_String &other) { _init(); set(other.ptr, other.len); }
591
592 ~LL_String() { if (ptr) { delete[] ptr; ptr = 0; } len = cap = 0; }
593
594 void set(const char *s) { set(s, s ? (int) strlen(s) : 0); }
595
596 void set(const char *s, int n) {
597 if (!s || n <= 0) { _ensure(0); ptr[0] = '\0'; len = 0; return; }
598 if (ptr && s >= ptr && s <= ptr + len) {
599 const int off = (int) (s - ptr);
600 memmove(ptr, ptr + off, (size_t) n);
601 }
602 else {
603 _ensure(n);
604 memmove(ptr, s, (size_t) n);
605 }
606 ptr[n] = '\0';
607 len = n;
608 }
609
610 LL_String concat(const char *s1, const char *s2) {
611 LL_String r;
612 const int n1 = s1 ? (int) strlen(s1) : 0;
613 const int n2 = s2 ? (int) strlen(s2) : 0;
614 r._ensure(n1 + n2);
615 if (n1) memmove(r.ptr, s1, (size_t) n1);
616 if (n2) memmove(r.ptr + n1, s2, (size_t) n2);
617 r.ptr[n1 + n2] = '\0';
618 r.len = n1 + n2;
619 return r;
620 }
621
622 LL_String &append(const char *s1) {
623 if (s1 && *s1) {
624 const int n1 = (int) strlen(s1);
625 const bool self = (ptr && s1 >= ptr && s1 <= ptr + len);
626 const int off = self ? (int) (s1 - ptr) : 0;
627 _ensure(len + n1);
628 memmove(ptr + len, self ? ptr + off : s1, (size_t) n1);
629 len += n1;
630 ptr[len] = '\0';
631 }
632 return *this;
633 }
634
637 LL_String &append(char c) {
638 _ensure(len + 1);
639 ptr[len++] = c;
640 ptr[len] = '\0';
641 return *this;
642 }
643
644 LL_String substring(int start, int len_) { return LL_String(&(ptr[start]), len_); }
645
646 LL_String &operator=(const char *other) { set(other); return *this; }
647 LL_String &operator=(const LL_String &other) { if (this != &other) set(other.ptr, other.len); return *this; }
648
649 bool operator==(const char *other) { return strcmp(ptr, other) == 0; }
650 bool operator!=(const char *other) { return strcmp(ptr, other) != 0; }
651 bool operator==(const LL_String &other) { return strcmp(ptr, other.ptr) == 0; }
652 bool operator!=(const LL_String &other) { return strcmp(ptr, other.ptr) != 0; }
653 char &operator[](int ix) { return ptr[ix]; }
654
655 LL_String operator+(const char *other) { return concat(ptr, other); }
656 LL_String operator+(const LL_String &other) { return concat(ptr, other.ptr); }
657 LL_String operator+(char c) { char cstr[2]; cstr[0] = c; cstr[1] = '\0'; return concat(ptr, cstr); }
658
659 LL_String &operator+=(const char *other) { return append(other); }
660 LL_String &operator+=(const LL_String &other) { return append(other.ptr); }
661 LL_String &operator+=(char c) { return append(c); }
662
663 char *c_str() const { return ptr; }
664 int length() const { return len; }
665
666private:
667 void _init() { ptr = 0; len = 0; cap = 0; }
668
671 void _ensure(int need) {
672 if (cap >= need + 1) return;
673 int ncap = cap ? cap : 16;
674 while (ncap < need + 1) ncap += ncap;
676 char *np = new char[ncap];
677 if (ptr) { memmove(np, ptr, (size_t) len + 1); delete[] ptr; }
678 else { np[0] = '\0'; }
679 ptr = np;
680 cap = ncap;
681 }
682
683 char *ptr;
684 int len;
685 int cap;
686};

1073inline int ll_u8_seqlen(unsigned char b0)
1074{
1075 if (b0 < 0x80) return 1;
1076 if ((b0 & 0xE0) == 0xC0) return 2;
1077 if ((b0 & 0xF0) == 0xE0) return 3;
1078 if ((b0 & 0xF8) == 0xF0) return 4;
1079 return 1;
1080}
1081
1084inline ll_codepoint_t ll_u8_decode(const char *p, int *nb)
1085{
1086 unsigned char b0 = (unsigned char) p[0];
1087 int n = ll_u8_seqlen(b0);
1088 if (nb) *nb = n;
1089 if (n == 1) return (ll_codepoint_t) b0;
1090 ll_codepoint_t cp = b0 & (0xFF >> (n + 1));
1091 for (int i = 1; i < n; i++) {
1092 unsigned char c = (unsigned char) p[i];
1093 if ((c & 0xC0) != 0x80) { if (nb) *nb = 1; return (ll_codepoint_t) b0; }
1094 cp = (cp << 6) | (c & 0x3F);
1095 }
1096 return cp;
1097}

1100inline int ll_u8_encode(ll_codepoint_t cp, char *out)
1101{
1102 if (cp < 0x80) { out[0] = (char) cp; return 1; }
1103 if (cp < 0x800) { out[0] = (char)(0xC0 | (cp >> 6)); out[1] = (char)(0x80 | (cp & 0x3F)); return 2; }
1104 if (cp < 0x10000) { out[0] = (char)(0xE0 | (cp >> 12)); out[1] = (char)(0x80 | ((cp >> 6) & 0x3F));
1105 out[2] = (char)(0x80 | (cp & 0x3F)); return 3; }
1106 out[0] = (char)(0xF0 | (cp >> 18)); out[1] = (char)(0x80 | ((cp >> 12) & 0x3F));
1107 out[2] = (char)(0x80 | ((cp >> 6) & 0x3F)); out[3] = (char)(0x80 | (cp & 0x3F)); return 4;
1108}

1170 char *dec(Word_t n) {
1171 char *ptr = slot_end();
1172
1173 *ptr = '\0';
1174 do {
1175 Word_t d = n % 10;
1176 n /= 10;
1177 char c = d + '0';
1178 *--ptr = c;
1179 } while (n);
1180 return ptr;
1181 }

1185 if (n == INT32_MIN) return dec((LL_int64) n);
1186 char *ptr = dec(-n);
1187 *--ptr = '-';
